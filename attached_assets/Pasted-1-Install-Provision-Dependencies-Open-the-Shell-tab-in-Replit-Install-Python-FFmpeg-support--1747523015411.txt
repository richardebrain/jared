3. Add the EduTalk Router
Create backend/routers/edutalk.py:

python
Copy
Edit
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session
from backend import models
from backend.main import get_db

router = APIRouter(prefix="/api/edutalk", tags=["edutalk"])

@router.get("/feed")
def get_feed(cursor: int = 0, limit: int = 10, db: Session = Depends(get_db)):
    items = (
      db.query(models.Snippet)
        .filter(models.Snippet.id > cursor)
        .order_by(models.Snippet.id)
        .limit(limit)
        .all()
    )
    return {"snippets": items}

4. Build the Ingest Script
Create backend/ingest_snippets.py to pull & clip videos:

Copy
Edit
import os, json, subprocess
from googleapiclient.discovery import build
from sqlalchemy.orm import Session
from backend import models
from backend.main import SessionLocal

YOUTUBE_API_KEY = os.getenv("YOUTUBE_API_KEY")
youtube = build("youtube", "v3", developerKey=YOUTUBE_API_KEY)

def fetch_video_ids(query, max_results=5):
    res = youtube.search().list(q=query, part="id,snippet", type="video", maxResults=max_results).execute()
    return [(item["id"]["videoId"], item["snippet"]["title"]) for item in res["items"]]

def download_and_clip(video_id, title, start, duration, out_path):
    url = f"https://www.youtube.com/watch?v={video_id}"
    # download full video
    subprocess.run(["youtube-dl", "-f", "mp4", "-o", "full.%(ext)s", url])
    # clip snippet
    subprocess.run([
      "ffmpeg", "-ss", str(start), "-t", str(duration),
      "-i", "full.mp4", "-c", "copy", out_path
    ])

def ingest(query):
    db: Session = SessionLocal()
    for vid, title in fetch_video_ids(query):
        clip_file = f"snippets/{vid}.mp4"
        download_and_clip(vid, title, start=30, duration=20, out_path=clip_file)
        video_url, thumb = upload_to_stream(clip_file, title)
        snippet = models.Snippet(
          title=title,
          video_url=video_url,
          thumbnail_url=thumb,
          source_url=f"https://youtu.be/{vid}",

5. Add the Infinite-Scroll Component
In your front-end (e.g. React src/components/EduTalkFeed.js):

jsx
Copy
Edit
import React, { useState, useEffect } from "react";
import axios from "axios";
import InfiniteScroll from "react-infinite-scroll-component";

export default function EduTalkFeed() {
  const [snippets, setSnippets] = useState([]);
  const [cursor, setCursor] = useState(0);

  const fetchMore = async () => {
    const res = await axios.get("/api/edutalk/feed", {
      params: { cursor, limit: 5 },
    });
    if (res.data.snippets.length) {
      setSnippets([...snippets, ...res.data.snippets]);
      setCursor(res.data.snippets.slice(-1)[0].id);
    }
  };

  useEffect(() => { fetchMore() }, []);

  return (
    <InfiniteScroll
      dataLength={snippets.length}
      next={fetchMore}
      hasMore={true}
      loader={<h4>Loading…</h4>}
    >
      {snippets.map((s) => (
        <div key={s.id} className="video-card">
          <video
            src={s.video_url}
            poster={s.thumbnail_url}
            controls
            autoPlay
            muted
            loop
            style={{ width: "100%" }}
          />
          <p>{s.title}</p>
        </div>
      ))}
    </InfiniteScroll>
  );
}
Import and render <EduTalkFeed /> in your app’s main page.
