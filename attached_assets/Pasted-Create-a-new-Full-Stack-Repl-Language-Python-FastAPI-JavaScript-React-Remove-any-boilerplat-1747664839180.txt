import pandas as pd
from .database import SessionLocal, engine, Base
from .models import Question

Base.metadata.create_all(bind=engine)
df = pd.read_csv("../data/ece_master_database_ready.csv")
db = SessionLocal()
for _, row in df.iterrows():
    q = Question(**row.to_dict())
    db.add(q)
db.commit()
db.close()

Build the Assessment UI

Create src/Assessment.js:

import React, { useState, useEffect } from "react";
import axios from "axios";

function Assessment() {
  const [step, setStep] = useState("start"); 
  const [history, setHistory] = useState([]);

  useEffect(() => {
    if (step==="start") {
      axios.post("/api/assessment/start").then(()=>setStep("next"));
    }
  }, [step]);

  const getNext = () => {
    axios.get("/api/assessment/next", { params:{ history } })
      .then(res => setStep(res.data));
  };

  const submit = (questionId, choice) => {
    const correct = choice===step.answer;
    setHistory([...history, { question_id: questionId, correct }]);
    setStep("next");
  };

  if (step==="next") { getNext(); return <div>Loading question…</div>; }
  if (step.id) {
    return (
      <div>
        <h2>{step.question}</h2>
        {Object.entries(step.options).map(([key, txt])=>(
          <button key={key} onClick={()=>submit(step.id, key)}>{key}. {txt}</button>
        ))}
      </div>
    );
  }
  return <div>Starting assessment…</div>;
}
export default Assessment;
In src/App.js, render <Assessment />.
