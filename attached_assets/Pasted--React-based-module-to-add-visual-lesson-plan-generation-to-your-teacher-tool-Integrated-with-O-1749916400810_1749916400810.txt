// React-based module to add visual lesson plan generation to your teacher tool
// Integrated with OpenAI DALL·E image generation via fetch API

import React, { useState } from 'react';
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";
import { Textarea } from "@/components/ui/textarea";
import { Select, SelectTrigger, SelectItem, SelectContent } from "@/components/ui/select";

const MAX_IMAGES_PER_DAY = 5;

const generateImageFromPrompt = async (prompt) => {
  const response = await fetch("https://api.openai.com/v1/images/generations", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer YOUR_OPENAI_API_KEY` // Replace with your actual API key securely
    },
    body: JSON.stringify({
      prompt,
      n: 1,
      size: "1024x1024"
    })
  });

  const data = await response.json();
  return data?.data?.[0]?.url || null;
};

export default function LessonPlanVisualizer() {
  const [lessonText, setLessonText] = useState("");
  const [imageURL, setImageURL] = useState(null);
  const [imageCount, setImageCount] = useState(0);
  const [generating, setGenerating] = useState(false);
  const [style, setStyle] = useState("whimsical");
  const [purpose, setPurpose] = useState("outline");
  const [audience, setAudience] = useState("parents");

  const handleGenerate = async () => {
    if (imageCount >= MAX_IMAGES_PER_DAY || !lessonText.trim()) return;
    setGenerating(true);

    const promptBase = `Create a ${style} ${purpose === "outline" ? "infographic-style visual of the entire lesson plan" : "set of images to support lesson plan sections"} for ${audience}. Content: ${lessonText}`;
    const img = await generateImageFromPrompt(promptBase);
    setImageURL(img);
    setImageCount(prev => prev + 1);
    setGenerating(false);
  };

  return (
    <Card className="p-4">
      <CardContent>
        <h2 className="text-xl font-bold mb-2">Visualize Your Lesson Plan</h2>

        <Textarea
          placeholder="Paste your lesson plan here..."
          value={lessonText}
          onChange={e => setLessonText(e.target.value)}
          rows={6}
        />

        <div className="grid grid-cols-3 gap-4 mt-4">
          <div>
            <label className="block font-medium">Visual Type</label>
            <Select onValueChange={setPurpose} defaultValue="outline">
              <SelectTrigger>{purpose}</SelectTrigger>
              <SelectContent>
                <SelectItem value="outline">Infographic-style Visual</SelectItem>
                <SelectItem value="sections">Images for Each Section</SelectItem>
              </SelectContent>
            </Select>
          </div>

          <div>
            <label className="block font-medium">Style</label>
            <Select onValueChange={setStyle} defaultValue="whimsical">
              <SelectTrigger>{style}</SelectTrigger>
              <SelectContent>
                <SelectItem value="whimsical">Playful/Whimsical</SelectItem>
                <SelectItem value="modern">Clean/Modern</SelectItem>
                <SelectItem value="storybook">Storybook/Cartoon</SelectItem>
                <SelectItem value="realistic">Photographic/Realistic</SelectItem>
              </SelectContent>
            </Select>
          </div>

          <div>
            <label className="block font-medium">Audience</label>
            <Select onValueChange={setAudience} defaultValue="parents">
              <SelectTrigger>{audience}</SelectTrigger>
              <SelectContent>
                <SelectItem value="training">Teacher Training</SelectItem>
                <SelectItem value="parents">Parent Handout</SelectItem>
                <SelectItem value="app">App/Classroom Visual</SelectItem>
              </SelectContent>
            </Select>
          </div>
        </div>

        <Button
          className="mt-6"
          onClick={handleGenerate}
          disabled={generating || imageCount >= MAX_IMAGES_PER_DAY}
        >
          {generating ? "Generating..." : `Generate Visual (${MAX_IMAGES_PER_DAY - imageCount} left)`}
        </Button>

        {imageURL && (
          <div className="mt-6">
            <h3 className="text-lg font-semibold">Generated Visual</h3>
            <img src={imageURL} alt="Generated Lesson Plan Visual" className="mt-2 rounded shadow" />
          </div>
        )}
      </CardContent>
    </Card>
  );
}
