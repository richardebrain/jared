Add a secret named GOAPI_KEY (value = the X-API-Key you got from GoAPI.ai).

This ensures your key never lands in source control.

4. Add a Protected Route for Song Generation
In backend/server.js (or equivalent):

js
Copy
Edit
import axios from "axios";

// helper to get “YYYY-WW” week code
function getWeekCode() {
  const d = new Date();
  const year = d.getUTCFullYear();
  const week = String(Math.ceil((((d - new Date(year,0,1)) / 86400000) + new Date(year,0,1).getUTCDay()+1)/7)).padStart(2,"0");
  return `${year}-${week}`;
}

app.post("/api/director/song", authenticateDirector, async (req, res) => {
  const { prompt } = req.body;
  const user = await db.getUser(req.user.id);

  // 1) Check weekly limit
  const currentWeek = getWeekCode();
  if (user.lastRequestWeekStart !== currentWeek) {
    user.songRequestsThisWeek = 0;
    user.lastRequestWeekStart = currentWeek;
  }
  if (user.songRequestsThisWeek >= 1) {
    return res.status(429).json({ error: "Weekly song limit reached" });
  }

  // 2) Basic content filter (no profanity)
  const badWords = ["badword1","badword2"]; // expand as needed
  for (let w of badWords) {
    if (prompt.toLowerCase().includes(w)) {
      return res.status(400).json({ error: "Please remove inappropriate language" });
    }
  }

  // 3) Call GoAPI Suno endpoint
  try {
    const goRes = await axios.post(
      "https://api.goapi.ai/api/suno/v1/music",
      {
        custom_mode: false,
        input: { gpt_description_prompt: prompt }
      },
      { headers: { "X-API-Key": process.env.GOAPI_KEY } }
    );

    // 4) Update user usage
    user.songRequestsThisWeek++;
    await db.saveUser(user);

    // 5) Stream back the audio URL or base64
    res.json({ audioUrl: goRes.data.audio_url });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: "Song generation failed, please try again." });
  }
});
authenticateDirector middleware ensures only logged-in directors can call this.

db.getUser / db.saveUser represent your user storage logic.

5. Frontend: Director Toolkit UI
In your React DirectorToolkit component:

jsx
Copy
Edit
import { useState, useEffect } from "react";
import axios from "axios";

export default function DirectorToolkit() {
  const [prompt, setPrompt] = useState("");
  const [audioUrl, setAudioUrl] = useState(null);
  const [error, setError] = useState("");
  const [usedThisWeek, setUsedThisWeek] = useState(false);

  useEffect(() => {
    // on mount, fetch /api/director/status to see if they’ve used their song
    axios.get("/api/director/status").then(res => {
      setUsedThisWeek(res.data.usedThisWeek);
    });
  }, []);

  async function generateSong() {
    setError("");
    try {
      const res = await axios.post("/api/director/song", { prompt });
      setAudioUrl(res.data.audioUrl);
      setUsedThisWeek(true);
    } catch (e) {
      setError(e.response?.data?.error || "Unknown error");
    }
  }

  return (
    <div className="toolkit">
      <h2>Director’s Song Studio</h2>
      {usedThisWeek ? (
        <p>You’ve used your song for this week. Come back next week for a new tune!</p>
      ) : (
        <>
          <textarea
            placeholder="Enter song theme, child’s name, age, or kit..."
            value={prompt}
            onChange={e => setPrompt(e.target.value)}
            disabled={usedThisWeek}
          />
          <button onClick={generateSong} disabled={!prompt}>
            Generate Song
          </button>
          {error && <p className="error">{error}</p>}
        </>
      )}
      {audioUrl && (
        <audio controls src={audioUrl} className="mt-4">
          Your browser does not support audio playback.
        </audio>
      )}
    </div>
  );
}
/api/director/status returns { usedThisWeek: true|false } by checking the same user record.

Disable the form once a song’s been generated.

6. Testing & Deployment
Local Test:

Sign in as a director, hit the toolkit, generate a song, verify audio playback.

Edge Cases:

Try generating twice in one week → should get “Weekly song limit reached.”

Submit a prompt with a “bad word” → should get the profanity error.

Deploy:

Ensure GOAPI_KEY is in your Replit secrets.

Push and test on your live Repl; directors will now see a one-song-per-week, customizable, kid-friendly song generator.
